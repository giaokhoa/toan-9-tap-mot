\section{Khái niệm phương trình\\và hệ hai phương trình\\bậc nhất hai ẩn}

\begin{lesson}
    \subsection{Kiến thức trọng tâm}

    \textbf{Mở đầu.} Xét bài toán cổ sau:

    Quýt, cam mười bảy quả tươi\\
    Đem chia cho một trăm người cùng vui.\\
    Chia ba mỗi quả quýt rồi,\\
    Còn cam, mỗi quả chia mười vừa xinh.\\
    Trăm người, trăm miếng ngọt lành.\\
    Quýt, cam mỗi loại tính rành là bao?

    Trong bài toán này có hai đại lượng chưa biết (số cam và số quýt). Vậy ta có thể giải bài toán đó tương tự ``giải bài toán bằng cách lập phương trình'' được hay không? Để trả lời câu hỏi này, trước hết chúng ta cần tìm hiểu về phương trình và hệ hai phương trình bậc nhất hai ẩn.

    \subsubsection{Phương trình bậc nhất hai ẩn}

    Gọi $x$ là số cam, $y$ là số quýt (với $x$, $y$ nguyên dương).

    \textbf{Hoạt động 1.} Câu ``Quýt, cam mười bảy quả tươi'' có nghĩa là tổng số cam và số quýt là $17$. Hãy viết hệ thức với hai biến $x$ và $y$ biểu thị giả thiết này.

    \answerline

    \textbf{Hoạt động 2.} Tương tự, hãy viết hệ thức với hai biến $x$ và $y$ biểu thị giả thiết cho bởi các câu thơ thứ ba, thứ tư và thứ năm.

    \answerline
    \answerline

    Các hệ thức nhận được trong HĐ1, HĐ2 là những ví dụ về phương trình bậc nhất hai ẩn.

    \begin{keyconcept}
        Phương trình bậc nhất hai ẩn $x$ và $y$ là hệ thức dạng
        \[
            ax+by=c \tag{1}
        \]
        trong đó $a$, $b$ và $c$ là các số đã biết ($a\ne0$ hoặc $b\ne0$).

        Nếu tại $x=x_0$ và $y=y_0$ ta có $ax_0+by_0=c$ là một khẳng định đúng thì cặp số $(x_0;y_0)$ được gọi là một nghiệm của phương trình (1).
    \end{keyconcept}

    \textbf{Luyện tập 1.} Hãy viết một phương trình bậc nhất hai ẩn và chỉ ra một nghiệm của nó.

    \answerline
    \answerline

    \begin{keyconcept}
        Mỗi phương trình bậc nhất hai ẩn đều có vô số nghiệm.
    \end{keyconcept}

    \textbf{Luyện tập 2.} Viết nghiệm và biểu diễn hình học tất cả các nghiệm của mỗi phương trình bậc nhất hai ẩn sau:
    \begin{enumerate}[label=\alph*)]
        \item $2x-3y=5$;
        \item $0x+y=-3$;
        \item $x+0y=-2$.
    \end{enumerate}

    \answerline
    \answerline
    \answerline

    \subsubsection{Hệ hai phương trình bậc nhất hai ẩn}

    Trong HĐ1 và HĐ2, bài toán mở đầu dẫn đến hai phương trình bậc nhất hai ẩn là $x+y=17$ và $10x+3y=100$. Để giải bài toán, ta cần tìm các giá trị của $x$ và $y$ đồng thời thỏa mãn cả hai phương trình này.

    \begin{keyconcept}
        Một cặp gồm hai phương trình bậc nhất hai ẩn $ax+by=c$ và $a'x+b'y=c'$ được gọi là một hệ hai phương trình bậc nhất hai ẩn. Ta thường viết hệ phương trình đó dưới dạng
        \[
            \left\{
            \begin{aligned}
                ax+by&=c,\\
                a'x+b'y&=c'.
            \end{aligned}
            \right. \tag{*}
        \]

        Mỗi cặp số $(x_0;y_0)$ được gọi là một nghiệm của hệ (*) nếu nó đồng thời là nghiệm của cả hai phương trình của hệ (*).

        Mỗi nghiệm của hệ (*) chính là một nghiệm chung của hai phương trình của hệ (*).
    \end{keyconcept}

    \textbf{Luyện tập 3.} Trong hai cặp số $(0;-2)$ và $(2;-1)$, cặp số nào là nghiệm của hệ phương trình
    \[
        \left\{
        \begin{aligned}
            x-2y&=4,\\
            4x+3y&=5
        \end{aligned}
        \right. ?
    \]

    \answerline
    \answerline

    \textbf{Vận dụng.} Xét bài toán cổ trong Tình huống mở đầu. Gọi $x$ là số cam, $y$ là số quýt cần tính ($x,y\in\mathbb{N}^{*}$), ta có hệ phương trình bậc nhất hai ẩn sau:
    \[
        \left\{
        \begin{aligned}
            x+y&=17,\\
            10x+3y&=100.
        \end{aligned}
        \right.
    \]
    Trong hai cặp số $(10;7)$ và $(7;10)$, cặp số nào là nghiệm của hệ phương trình trên? Từ đó cho biết một phương án về số cam và số quýt thỏa mãn yêu cầu của bài toán cổ.

    \answerline
    \answerline
\end{lesson}
