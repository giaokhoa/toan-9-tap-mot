% Hỗ trợ tiếng Việt theo tùy chọn `vietnamese` của document class.
\usepackage{babel}

% Quản lý phông OpenType/TrueType khi biên dịch bằng XeLaTeX.
\usepackage{fontspec}

% Công thức và ký hiệu toán học.
\usepackage{amsmath, amssymb}

% Chèn và quản lý đường dẫn hình ảnh.
\usepackage{graphicx}

% Chia nội dung thành nhiều cột.
\usepackage{multicol}

% Thiết lập khổ giấy, lề và vùng đầu/chân trang.
\usepackage{geometry}

% Điều chỉnh giãn dòng.
\usepackage{setspace}

% Định nghĩa và sử dụng màu.
\usepackage{xcolor}

% Vẽ đồ họa và các thành phần trang bằng TikZ.
\usepackage{tikz}

% Tùy biến đầu trang và chân trang.
\usepackage{fancyhdr}

% Tùy biến hình thức các cấp tiêu đề; `explicit` cho phép dùng #1.
\usepackage[explicit]{titlesec}

% Tùy biến danh sách itemize, enumerate và danh sách riêng.
\usepackage{enumitem}

% Trình bày các lựa chọn hoặc bài tập trên nhiều cột.
\usepackage{tasks}

% Tạo hộp nội dung có giao diện tùy biến và hỗ trợ ngắt trang.
\usepackage[skins, breakable]{tcolorbox}

% Tạo bảng với cú pháp khóa--giá trị.
\usepackage{tabularray}
